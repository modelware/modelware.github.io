queries/trace.rq
SELECT ?requirement ?satisfiedBy ?mass ?verdict
WHERE {
  ?s a sysml:Satisfy ;
     sysml:satisfiedRequirement ?r ;
     sysml:satisfyingFeature ?f .
  BIND(eval:name(?r)         AS ?requirement)
  BIND(eval:name(?f)         AS ?satisfiedBy)
  BIND(eval:text("mass", ?f) AS ?mass)
  BIND(eval:verdict(?r)      AS ?verdict)
}
